% ---- centred grid of small tiles (drawn from the current origin) ----
% #1=lastCol (cols-1)  #2=lastRow (rows-1)  #3=colour  #4=step  #5=cell
\providecommand{\tgtiles}[5]{%
  \foreach \tx in {0,...,#1}{%
    \foreach \ty in {0,...,#2}{%
      \fill[#3] (\tx*#4 - #1*#4/2 - #5/2, \ty*#4 - #2*#4/2 - #5/2)
                rectangle +(#5,#5);}}%
}

    \caption{Training infrastructure used for the TG-1 smoke tests
    (\texttt{scripts/smoke.py}).  The on-device loop (dashed box) holds a single
    batched \texttt{VSSVecEnv} that advances 256--1024 matches in parallel through
    $\mathrm{jit}(\mathrm{vmap}(\texttt{step}))$ on the JAX device.  Each step
    exports a $\mathbb{R}^{46}$ observation to the MLP policy, which returns the
    $\mathbb{R}^{6}$ joint wheel-speed action; collected trajectories cross the
    Gymnasium\,$\rightleftarrows$\,SB3 adapter to the PPO learner (PyTorch, CPU),
    whose updated parameters $\theta$ flow back to the policy.  Episode resets are
    driven by the pluggable \texttt{InitMode} (kickoff or random, the future
    insertion point for the PLR sampler), and MLflow logs per-generation reward,
    episode length, throughput and checkpoints.}
    \label{fig:training_infra}
